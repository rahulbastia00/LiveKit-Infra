Install Docker CLI inside the running Jenkins container:
docker exec -u 0 jenkins apt-get update
docker exec -u 0 jenkins apt-get install -y docker.io

Get the initial admin password:
docker exec jenkins cat /var/jenkins_home/secrets/initialAdminPassword

Run the containerized Trivy command directly against the test image you just built:

Bash
# 1. Human-readable table scan (informational)
docker run --rm \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v trivy-cache:/root/.cache/ \
  aquasec/trivy:latest image \
  --severity HIGH,CRITICAL \
  --format table \
  test-livekit-client:local

# 2. Gate check test (returns exit code 1 if unpatched HIGH/CRITICAL issues exist)
docker run --rm \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v trivy-cache:/root/.cache/ \
  aquasec/trivy:latest image \
  --exit-code 1 \
  --severity HIGH,CRITICAL \
  --ignore-unfixed \
  test-livekit-client:local
